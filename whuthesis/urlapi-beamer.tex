\section{项目定位}

\begin{frame}{项目定位}
  \begin{block}{一句话概括}
    urlAPI 是一个 Go + Vue 的 URL 化 API 服务：把文本生成、图片生成、随机图片、网页缩略图和 OpenAI 兼容模型接口封装成可直接调用的 Web 服务，并提供后台管理能力。
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\linewidth}
      \textbf{主要能力}
      \begin{itemize}
        \item 文字生成并渲染为 PNG 图片
        \item 图片生成并通过下载接口中转
        \item GitHub/Gitee 仓库随机图片
        \item Bilibili、YouTube、arXiv、GitHub 等网页信息卡片
        \item OpenAI-compatible \code{/v1} 接口
      \end{itemize}
    \end{column}
    \begin{column}{0.48\linewidth}
      \textbf{管理能力}
      \begin{itemize}
        \item 后台登录、任务查看与筛选
        \item 供应商、模型、参数和 Prompt 配置
        \item Referer、IP、Prompt、功能开关安全策略
        \item API Key 创建、配额、IP 白名单和启停
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

    Router 装载安全中间件 $\rightarrow$ Handler 读取参数 $\rightarrow$ op 执行业务与任务缓存 $\rightarrow$ database 记录任务 $\rightarrow$ JSON 返回或 302 跳转到结果 URL。

\section{功能展示}

\begin{frame}[fragile]{访问方式一：URL + Referer 白名单}
  \textbf{面向博客、页面和简单自动化：把结果当作图片或 JSON 使用。}

  \begin{verbatim}
curl -L "https://api.example.com/txt?prompt=poem&format=json"
curl -L "https://api.example.com/img?prompt=cat&api=openai&size=1024x1024"
curl -L "https://api.example.com/rand?api=github&user=me&repo=images"
curl -L "https://api.example.com/web?img=https://github.com/owner/repo"
  \end{verbatim}

  \begin{itemize}
    \item 默认非 JSON 请求返回 \code{302}，浏览器或 \code{curl -L} 会跟随到图片地址。
    \item \code{format=json} 返回结构化结果，便于脚本调试。
    \item 公开接口经过 Referer 白名单、频率限制、功能开关、Prompt 通配符等检查。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{访问方式二：curl + API Key}
  \textbf{面向开发者：通过 OpenAI 兼容 \code{/v1} 接口接入。}

  \begin{verbatim}
curl https://api.example.com/v1/chat/completions \
  -H "Authorization: Bearer sk-..." \
  -H "X-Provider: deepseek" \
  -H "Content-Type: application/json" \
  -d '{"model":"deepseek-chat","messages":[{"role":"user","content":"Hi"}]}'
  \end{verbatim}

  \begin{itemize}
    \item API Key 可从 \code{Authorization: Bearer}、\code{X-API-Key} 或 \code{api\_key} 查询参数读取。
    \item 支持 Key 启停、角色、日/月配额、过期时间和 IP 白名单。
    \item \code{X-Provider} 可切换供应商；不传时使用后台默认文字供应商。
  \end{itemize}
\end{frame}

        \item \code{APIKeyUsage} 日志记录每次 API Key 调用的端点、IP、状态码和耗时。
        \item \code{GetUsageStats} 支持按 API Key 和起止时间查询调用量。
      \end{itemize}
    \end{column}
    \begin{column}{0.48\linewidth}
      \textbf{配额与并发统计}
      \begin{itemize}
        \item API Key 记录 \code{usage\_day} 和 \code{usage\_month}，实时追踪日/月使用量。
        \item \code{CheckQuota} 校验调用是否超出日/月配额上限。
        \item \code{TaskCounter} 按上游 API 统计当前并发任务数，防止打满。
        \item \code{ResetDailyUsage/ResetMonthlyUsage} 自动重置过期日/月计数。

  \begin{block}{统计维度汇总}

\section{特性设计}

\begin{frame}{特性设计：执行队列}
  \begin{block}{核心数据结构}
    \code{TaskQueue} 以 \code{Type + Size + Target + API} 作为去重键；\code{TaskCounter} 按 API 统计并发任务数。
  \end{block}

  \begin{itemize}
    \item 同一任务正在运行时，后续请求等待结果，避免重复调用上游 API。
    \item 成功任务在缓存时间内复用结果，当前任务标记 \code{Temp=Yes}。
    \item 过期或失败任务会清理旧临时图片，重新执行。
    \item 每个 API 的并发执行数受控，避免上游接口或本机资源被打满。
    \item 任务结束后会校验 PNG 文件有效性，异常时返回空图并记录失败。
  \end{itemize}
\end{frame}

\begin{frame}{特性设计：图片缓存能力}
  \begin{block}{缓存目标}
    对文字生成图、AI 生成图和网页缩略图进行结果复用，减少重复调用上游 API、重复绘图和重复下载。
  \end{block}

  \begin{itemize}
    \item 缓存键复用 \code{TaskQueueFilter}：按任务类型、尺寸、目标内容和 API 供应商区分结果。
    \item 生成成功后图片保存到 \code{assets/img/<uuid>.png}，对外统一返回 \code{/download?img=<uuid>}。
    \item 缓存过期时间来自后台配置：文字、图片、网页缩略图分别有独立 \code{cache\_minutes}。
    \item 命中缓存时不重新生成图片，而是复制历史任务结果并标记 \code{Temp=Yes}，仍可按需写入任务记录。
    \item 缓存过期、生成失败或 PNG 校验失败时会丢弃旧结果，重新执行并在失败时返回空图或 fallback。
    \item 临时图片目录在服务启动时重建，因此缓存是运行期缓存；持久化任务记录保留，但旧下载 URL 可能随重启失效。
  \end{itemize}
\end{frame}

\begin{frame}{特性设计：流控与风控}
  \begin{columns}[T]
    \begin{column}{0.48\linewidth}
      \textbf{流控}
      \begin{itemize}
        \item 公开 URL API 按 \code{Type + IP} 维护访问频率窗口。
        \item 0.25 秒内超过 10 次同类请求会被拒绝，避免脚本刷接口。
        \item API Key 支持日配额、月配额和使用量统计，超额返回 \code{rate\_limit\_error}。
        \item \code{TaskCounter} 按上游 API 统计并发，防止模型接口和本机资源被打满。
      \end{itemize}
    \end{column}
    \begin{column}{0.48\linewidth}
      \textbf{风控}
      \begin{itemize}
        \item 功能开关控制文字、图片、随机图、网页缩略图是否开放。
        \item Prompt 支持启用列表和通配符白名单，限制可生成内容范围。
        \item API 参数限制在受支持供应商集合内，阻断非法 provider 输入。
        \item 可选 HMAC 签名校验，使用时间戳降低重放风险。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{特性设计：白名单机制}
  \begin{table}
    \small
    \begin{tabularx}{\linewidth}{lX}
      \toprule
      白名单 & 项目实现与作用 \\

    \begin{column}{0.46\linewidth}

  \vspace{0.4cm}
  \begin{block}{失败处理}
    外部调用失败会落入任务记录，并按接口类型返回 JSON 错误、fallback 图片或 OpenAI 兼容错误结构。
  \end{block}
\end{frame}

\begin{frame}{特性设计：数据库版本匹配与更新}
  \begin{itemize}
    \item 旧版本使用 \code{settings.name + settings.value(JSON 字符串数组)}，业务和前端依赖魔法下标。
    \item 新设计拆出 \code{providers}、\code{service\_configs}、\code{prompts}、\code{config\_list\_items}、\code{app\_settings} 等结构化表。
    \item 启动时通过 \code{SettingsStore} 加载结构化配置，并用读写锁提供线程安全快照。
    \item 保存设置时先构建 V2 rows，在事务中替换相关表，再替换内存快照。
    \item 新 \code{fetchSettings/editSettings} 返回具名 DTO，敏感字段不明文回显，空值保留旧密钥。
    \item 迁移文档要求保留旧表一段时间，支持从旧数组配置安全迁移并避免破坏现有行为。
  \end{itemize}
\end{frame}
